cmake_minimum_required(VERSION 3.24)
project(cableloom_core_tests LANGUAGES CXX)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_compile_features(core_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(core_tests PRIVATE /W4 /WX)
else()
  target_compile_options(core_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME core COMMAND core_tests)
